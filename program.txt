function calculateSkillAnalysis(career, extractedSkills) {
  const requirements = roleRequirements[career] || {};

  const skills = Object.entries(requirements).map(
    ([name, required]) => {
      const current = Number(extractedSkills[name] || 0);

      const gap = Math.max(required - current, 0);

      let priority = "Ready";

      if (gap >= 30) {
        priority = "High";
      } else if (gap >= 15) {
        priority = "Medium";
      } else if (gap > 0) {
        priority = "Low";
      }

      return {
        name,
        required,
        score: current,
        gap,
        priority,
        available: current >= required,
      };
    }
  );

  const totalRequired = skills.reduce(
    (sum, skill) => sum + skill.required,
    0
  );

  const totalCurrent = skills.reduce(
    (sum, skill) =>
      sum + Math.min(skill.score, skill.required),
    0
  );

  const readiness = totalRequired
    ? Math.round((totalCurrent / totalRequired) * 100)
    : 0;

  const gaps = skills
    .filter((skill) => skill.gap > 0)
    .sort((a, b) => b.gap - a.gap);

  return {
    skills,
    gaps,
    readiness,
  };
}

function createDailyPlan(gaps, studyTime) {
  const hours =
    studyTime === "1 hour/day"
      ? 1
      : studyTime === "2 hours/day"
      ? 2
      : studyTime === "3 hours/day"
      ? 3
      : 4;

  const daysPerSkill = hours >= 3 ? 2 : 3;

  const plan = [];

  let day = 1;

  gaps.forEach((skill) => {
    for (let i = 0; i < daysPerSkill; i++) {
      const content = learningContent[skill.name];

      plan.push({
        day,
        skill: skill.name,
        title:
          content?.title ||
          `Learn ${skill.name}`,

        description:
          content?.description ||
          `Practice and improve your ${skill.name} skills.`,

        hours,
        status: day === 1 ? "Today" : "Upcoming",
      });

      day++;
    }
  });

  plan.push({
    day,
    skill: "Portfolio",
    title: "Build a Career Project",
    description:
      "Build one project using the skills learned in your roadmap.",
    hours,
    status: "Upcoming",
  });

  day++;

  plan.push({
    day,
    skill: "Interview",
    title: "Prepare for Interviews",
    description:
      "Practice technical questions, resume explanation and HR questions.",
    hours,
    status: "Upcoming",
  });

  return plan;
}

/* =========================================================
   LOADING
========================================================= */

function Loading({ onComplete }) {
  const [step, setStep] = useState(0);

  const steps = [
    "Reading career profile",
    "Extracting skills",
    "Checking target career requirements",
    "Finding skill gaps",
    "Prioritizing missing skills",
    "Building learning plan",
    "Preparing opportunity matches",
  ];

  useEffect(() => {

    const timer = window.setInterval(() => {

      setStep((current) => {

        if (current >= steps.length - 1) {

          window.clearInterval(timer);

          window.setTimeout(() => {
            onComplete();
          }, 700);

          return current;
        }

        return current + 1;
      });

    }, 600);

    return () =>
      window.clearInterval(timer);

  }, [onComplete]);

  return (
    <section className="loading-page">

      <div className="loading-content slide-up">

        <div className="ai-scanner">

          <div className="scanner-core">
            ✦
          </div>

          <div className="scanner-ring scanner-ring-one"></div>

          <div className="scanner-ring scanner-ring-two"></div>

        </div>

        <h1>
          AI is analyzing your
          <span className="gradient-text">
            {" "}career profile...
          </span>
        </h1>

        <p>
          Please wait while SkillPath AI builds
          your personalized career path.
        </p>

        <div className="analysis-steps">

          {steps.map((item, index) => (

            <div
              key={item}
              className={`analysis-step ${
                index <= step
                  ? "active"
                  : ""
              }`}
            >

              <span>
                {index <= step
                  ? "✓"
                  : "○"}
              </span>

              {item}

            </div>

          ))}

        </div>

      </div>
    </section>
  );
}

function App() {

  const [page, setPage] =
    useState("home");

  const [user, setUser] = useState(() => {

    try {

      const savedUser =
        localStorage.getItem(
          "skillpath_user"
        );

      return savedUser
        ? JSON.parse(savedUser)
        : null;

    } catch {

      return null;

    }

  });

  const [career, setCareer] =
    useState("ML Engineer");

  const [studyTime, setStudyTime] =
    useState("2 hours/day");

  const [resumeName, setResumeName] =
    useState("");

  const [mode, setMode] =
    useState("resume");

  const [analysis, setAnalysis] =
    useState({
      skills: [],
      gaps: [],
      readiness: 0,
    });

  const [plan, setPlan] =
    useState([]);

  const [
    selectedOpportunity,
    setSelectedOpportunity,
  ] = useState(null);


  const go = (nextPage) => {

    setPage(nextPage);

    window.scrollTo({
      top: 0,
      behavior: "smooth",
    });

  };


  const handleLogin = (loggedInUser) => {

    setUser(loggedInUser);

    go("form");

  };


  const handleLogout = () => {

    localStorage.removeItem(
      "skillpath_user"
    );

    setUser(null);

    go("home");

  };


  const handleAnalyze = (data) => {

    /*
      FRONTEND DEMO EXTRACTION

      Production architecture:

      Resume
        ↓
      FastAPI Backend
        ↓
      PyMuPDF / python-docx
        ↓
      Skill Extraction
        ↓
      Skill Normalization
        ↓
      SkillPath AI Analysis
    */

    const detectedSkills = {

      Python: 72,

      SQL: 68,

      Pandas: 70,

      "Machine Learning": 70,

      "Scikit-learn": 65,

    };

    let userSkills = detectedSkills;


    if (data.mode === "fresher") {

      userSkills = {};

    }


    const result =
      calculateSkillAnalysis(
        data.career,
        userSkills
      );


    const dailyPlan =
      createDailyPlan(
        result.gaps,
        data.studyTime
      );


    setCareer(data.career);

    setStudyTime(data.studyTime);

    setMode(data.mode);

    setResumeName(
      data.resume?.name || "No Resume"
    );

    setAnalysis(result);

    setPlan(dailyPlan);

    go("loading");

  };


  const matchingOpportunities =
    useMemo(
      () => demoOpportunities,
      []
    );


  return (
    <div className="app-shell">

      <Header

        onHome={() =>
          go("home")
        }

        onStart={() => {

          if (user) {

            go("form");

          } else {

            go("auth");

          }

        }}

        user={user}

        onLogin={() =>
          go("auth")
        }

        onLogout={handleLogout}

      />


      {page === "auth" && (

        <AuthPage
          onLogin={handleLogin}
        />

      )}


      {page === "home" && (
        <>

          <Landing
            onStart={() => {

              if (user) {

                go("form");

              } else {

                go("auth");

              }

            }}
          />

          <Footer />

        </>
      )}


      {page === "form" && (

        <AnalysisForm
          onAnalyze={handleAnalyze}
        />

      )}


      {page === "loading" && (

        <Loading
          onComplete={() =>
            go("dashboard")
          }
        />

      )}


      {page === "dashboard" && (

        <Dashboard
          career={career}
          analysis={analysis}
          resumeName={resumeName}
          mode={mode}
          onNext={() =>
            go("roadmap")
          }
        />

      )}


      {page === "roadmap" && (

        <Roadmap
          plan={plan}
          studyTime={studyTime}
          onNext={() =>
            go("opportunities")
          }
        />

      )}


      {page === "opportunities" && (

        <Opportunities
          opportunities={
            matchingOpportunities
          }

          onSelect={(item) => {

            setSelectedOpportunity(item);

            go("detail");

          }}

        />

      )}


      {page === "detail" &&
        selectedOpportunity && (

          <OpportunityDetail
            item={selectedOpportunity}
            onBack={() =>
              go("opportunities")
            }
          />

        )}

    </div>
  );
}
